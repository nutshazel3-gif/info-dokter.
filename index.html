Konfigurasi Apps Script Web App

Hubungkan endpoint Google Sheets ke portal web

Dapatkan URL ini dari menu Deploy > New deployment > Web app (Access: Anyone) pada Google Apps Script.

info Struktur Sheet yang didukung:

Google Spreadsheet memerlukan 3 sheet / tab dengan nama:
• DokterIzin: (nama_dokter, poli, tanggal_mulai, tanggal_selesai, kembali_praktik, keterangan, status)
• PerubahanKuota: (nama_dokter, poli, kuota_lama, kuota_baru, tanggal_efektif, keterangan, status)
• PerubahanJam: (nama_dokter, poli, jam_lama, jam_baru, tanggal_efektif, keterangan, status)

// ========================================================
// SISTEM IZIN PELAYANAN & INFO MEDIS RS. ISLAM BONTANG
// BACKEND GOOGLE APPS SCRIPT (Deploy as Web App)
// ========================================================

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    const dataIzin = getSheetData(ss, "DokterIzin");
    const dataKuota = getSheetData(ss, "PerubahanKuota");
    const dataJam = getSheetData(ss, "PerubahanJam");

    const payload = {
      status: "success",
      hospital: "RS. Islam Bontang",
      timestamp: new Date().toISOString(),
      counts: {
        dokter_izin: dataIzin.length,
        perubahan_kuota: dataKuota.length,
        perubahan_jam: dataJam.length
      },
      data: {
        dokter_izin: dataIzin,
        perubahan_kuota: dataKuota,
        perubahan_jam: dataJam
      }
    };

    return ContentService.createTextOutput(JSON.stringify(payload))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    const errorPayload = {
      status: "error",
      message: err.toString()
    };
    return ContentService.createTextOutput(JSON.stringify(errorPayload))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Utility: Ekstrak baris sheet menjadi JSON Array otomatis
function getSheetData(spreadsheet, sheetName) {
  const sheet = spreadsheet.getSheetByName(sheetName);
  if (!sheet) return [];
  
  const rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];

  const headers = rows[0].map(h => String(h).trim().toLowerCase().replace(/\s+/g, '_'));
  const results = [];

  for (let i = 1; i < rows.length; i++) {
    const row = rows[i];
    if (row.join("").trim() === "") continue; // skip row kosong
    const obj = {};
    headers.forEach((header, idx) => {
      let val = row[idx];
      if (val instanceof Date) {
        val = Utilities.formatDate(val, "Asia/Makassar", "yyyy-MM-dd");
      }
      obj[header] = val;
    });
    results.push(obj);
  }
  return results;
}
