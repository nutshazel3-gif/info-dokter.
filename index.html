Kode Google Apps Script (Code.gs)

Salin kode di bawah ini ke Extensions > Apps Script di Google Sheets RS. Islam Bontang, lalu klik Deploy > New Deployment > Web app (Who has access: Anyone): 

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 1. Sheet: Dokter Izin
    const sheetIzin = ss.getSheetByName('DokterIzin') || ss.getSheets()[0];
    const dataIzin = getSheetData(sheetIzin);
    
    // 2. Sheet: Perubahan Kuota
    const sheetKuota = ss.getSheetByName('PerubahanKuota');
    const dataKuota = sheetKuota ? getSheetData(sheetKuota) : [];
    
    // 3. Sheet: Perubahan Jam
    const sheetJam = ss.getSheetByName('PerubahanJam');
    const dataJam = sheetJam ? getSheetData(sheetJam) : [];
    
    const response = {
      status: 'success',
      timestamp: new Date().toISOString(),
      data: {
        dokterIzin: dataIzin,
        perubahanKuota: dataKuota,
        perubahanJam: dataJam
      }
    };
    
    return ContentService.createTextOutput(JSON.stringify(response))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getSheetData(sheet) {
  if (!sheet) return [];
  const rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return [];
  const headers = rows[0].map(h => String(h).trim());
  const data = [];
  for (let i = 1; i < rows.length; i++) {
    const row = rows[i];
    if (!row[0]) continue;
    let obj = {};
    headers.forEach((h, index) => {
      obj[h] = row[index];
    });
    data.push(obj);
  }
  return data;
}
Nama Kolom yang Dianjurkan di Google Sheets:
  • DokterIzin: nama_dokter, poli, tanggal_mulai, tanggal_selesai, kembali_praktik, keterangan, status, petugas
  • PerubahanKuota: nama_dokter, poli, kuota_semula, kuota_baru, tanggal_efektif, keterangan, status
  • PerubahanJam: nama_dokter, poli, jam_semula, jam_baru, tanggal_efektif, keterangan, status
